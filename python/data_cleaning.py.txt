import pandas as pd

# Load the raw January 2024 MOT dataset
file_path = r"D:\EDA PROJECT\Commercial Vehicle Reliability & Failure Pattern Analysis\datasets\raw_dataset.csv\test_result_202401.csv"

df = pd.read_csv(
    file_path,
    escapechar="\\"
)

# Remove the header-like row accidentally present in the dataset
df = df[df["test_date"] != "test_date"]

# Convert date columns into datetime format
df["test_date"] = pd.to_datetime(df["test_date"])
df["first_use_date"] = pd.to_datetime(df["first_use_date"], errors="coerce")
df["completed_date"] = pd.to_datetime(df["completed_date"], errors="coerce")

# Convert numeric columns into numeric format
df["test_mileage"] = pd.to_numeric(df["test_mileage"], errors="coerce")
df["cylinder_capacity"] = pd.to_numeric(df["cylinder_capacity"], errors="coerce")

# Remove exact duplicate records
df = df.drop_duplicates()

# Reset the index after cleaning
df = df.reset_index(drop=True)
